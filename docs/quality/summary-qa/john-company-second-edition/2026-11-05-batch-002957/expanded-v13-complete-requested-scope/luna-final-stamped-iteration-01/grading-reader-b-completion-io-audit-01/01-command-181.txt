    assert len(atoms)==len(parts),(qid,len(atoms),len(parts))
    reader=entry['reader_b_entry']
    texts=[reader['answer']]+[x for x in reader.get('case_conclusions',[]) if isinstance(x,str)]+[x['quote'] for x in reader.get('evidence',[]) if 'quote' in x]+[x['supported_fact_or_case_result'] for x in reader.get('clause_ledger',[]) if 'supported_fact_or_case_result' in x]

            assert any(quote in t for t in texts),(qid,quote)
