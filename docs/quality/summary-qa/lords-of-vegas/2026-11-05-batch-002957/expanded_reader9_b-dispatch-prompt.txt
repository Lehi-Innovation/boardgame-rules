```python
from copy import deepcopy

def render_answer(question_id, clause_refs, checked_result, operative_quotes, support, evidence, checked_inputs="not_applicable"):
    entry = dict(
        id=question_id,
        clauses=[dict(clause=c["clause"], ruling=deepcopy(checked_result), operative_unit_refs=c["operative_unit_refs"]) for c in clause_refs],
        operative_quotes=operative_quotes,
        answer=deepcopy(checked_result),
        support=support,
        evidence=evidence,
        case_check=dict(checked_inputs=checked_inputs, checked_result=deepcopy(checked_result)),
    )
    assert entry["answer"] == entry["case_check"]["checked_result"]
    assert all(c["ruling"] == entry["answer"] for c in entry["clauses"])
    return entry
```
